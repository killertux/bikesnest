{# Shared hidden-input mirrors for the three search.html forms (main search box, sort select, filter panel) so that submitting any one of them preserves the state the other two own. Each caller skips the field(s) it renders as a visible control, so no form ever submits a field twice. `skip_bbox` and `skip_origin` are what the main search box sets: typing a destination leaves browse mode and replaces any geolocated origin. Sort and filter forms keep both because they re-query the current search. #} {% macro hidden(form, skip_q, skip_sort, skip_filters, skip_bbox, skip_origin) %} {% if !skip_origin %} {% if let Some(lat) = form.lat %}{% endif %} {% if let Some(lon) = form.lon %}{% endif %} {% endif %} {% if !skip_q %}{% if !form.q.is_empty() %}{% endif %}{% endif %} {% if !skip_sort %}{% if !form.sort.is_empty() %}{% endif %}{% endif %} {% if !skip_bbox %}{% if !form.bbox.is_empty() %}{% endif %}{% endif %} {% if !skip_filters %} {% if let Some(r) = form.radius %}{% endif %} {% if !form.cost.is_empty() %}{% endif %} {% if !form.parking_type.is_empty() %}{% endif %} {% if !form.security.is_empty() %}{% endif %} {% if !form.open_now.is_empty() %}{% endif %} {% endif %} {% endmacro %} {# The "search this area" form: the map's current box plus whatever the viewer had narrowed down to — and deliberately no `q`/`lat`/`lon`, since browsing an area replaces searching around a destination. search.js writes the box into `#bbox-input` on every map move. #} {% macro browse(form) %} {% if !form.sort.is_empty() %}{% endif %} {% if !form.cost.is_empty() %}{% endif %} {% if !form.parking_type.is_empty() %}{% endif %} {% if !form.security.is_empty() %}{% endif %} {% if !form.open_now.is_empty() %}{% endif %} {% endmacro %}
